424 [[nodiscard]] auto checkout() -> resource_wrap<T> /* throw() */
425 {
426 {
427 std::scoped_lock<std::mutex> l(_poolLock);
428
429 if (!_pool.empty()) {
430 RunOnEnd roe([&]() { _pool.pop_front(); });
431
436 auto autoReturnResource = [this](T&& rsrc) {
437 this->checkin(std::move(rsrc));
438 };
439
440 return {std::move(_pool.front()), autoReturnResource};
441 // The pop_front() happens within this scope and
442 // within the lock!
443 }
444 } // scope end
445
446 throw std::runtime_error("Empty pool; add something first!");
447 }
